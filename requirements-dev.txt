# --- Asset pipeline ---------------------------------------------------------
# tools/build_atlas.py, gen_tiles.py, index_pack.py, check_layout.py and
# preview_map.py. Pillow does the compositing; numpy and scipy do the sprite
# segmentation (scipy.ndimage labels connected components when cutting a
# sprite out of a pack sheet).
#
# These also need, outside pip:
#   * node on PATH -- preview_map.py and check_layout.py read the real
#     office_map.js through `node -e` rather than keeping a second copy of the
#     layout that could drift
#   * the art source packs, which are NOT in this repo (see .gitignore and
#     STACK.md). The built atlas in frontend/assets/generated/ is committed, so
#     you only need the packs to regenerate it.
pillow==12.1.1
numpy==1.26.4
scipy==1.15.3
